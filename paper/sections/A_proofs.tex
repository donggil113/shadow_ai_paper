\subsection{Setup}

\begin{proof}[Proof of \cref{lem:msm}]
Fix $x$. By a standard reduction (see e.g.\ \citealp{dorn2023sharp}), for the
purpose of bounding functionals of $\Prb(Y\mid X, T=0)$ the latent $S$ may be
replaced by $Y$ itself: the marginal sensitivity model with arbitrary latent $S$
and the model stated with $e(x,y)=\Prb(T=1\mid X=x, Y=y)$ generate the same set
of admissible observed-data-compatible laws. So assume
\begin{equation}
  \Lambda^{-1}
  \;\le\;
  \frac{\odds\big(e(x)\big)}{\odds\big(e(x,y)\big)}
  \;\le\;\Lambda,
  \qquad y\in\{0,1\}.
  \label{eq:msm-y}
\end{equation}
By Bayes' rule,
\[
\begin{aligned}
  \odds\big(p_1(x)\big)
  &= \frac{\Prb(Y=1\mid x)\,e(x,1)}{\Prb(Y=0\mid x)\,e(x,0)}
  = \odds\big(p(x)\big)\cdot\frac{e(x,1)}{e(x,0)},\\
  \odds\big(p_0(x)\big)
  &= \odds\big(p(x)\big)\cdot\frac{1-e(x,1)}{1-e(x,0)} .
\end{aligned}
\]
Dividing,
\[
  \frac{\odds\big(p_0(x)\big)}{\odds\big(p_1(x)\big)}
  = \frac{(1-e(x,1))/e(x,1)}{(1-e(x,0))/e(x,0)}
  = \frac{\odds\big(e(x,0)\big)}{\odds\big(e(x,1)\big)} .
\]
By \eqref{eq:msm-y} each of $\odds(e(x,0))$, $\odds(e(x,1))$ lies in
$[\Lambda^{-1}\odds(e(x)),\,\Lambda\,\odds(e(x))]$, so their ratio lies in
$[\Lambda^{-2},\Lambda^{2}]$, which is \eqref{eq:dcsm} with $\Gamma=\Lambda^2$.
Attainment: take $e(x,1)$ at one end of its range and $e(x,0)$ at the other; both
choices are compatible with \eqref{eq:msm-y} and with the observed $e(x)$, so
every value in $[\Lambda^{-2},\Lambda^2]$ is realised.
\end{proof}

\begin{proof}[Proof of \cref{lem:box}]
Write $\eta_1(x)=\logit p_1(x)$. Given a tilt $a(\cdot)$, the induced regression
is \eqref{eq:tilt}, which is continuous and strictly increasing in $a(x)$ for
each $x$ with $e(x)<1$ (and constant, equal to $p_1(x)$, when $e(x)=1$). Hence
the image of $[\Gamma^{-1},\Gamma]$ under $a(x)\mapsto p^{a}(x)$ is exactly the
interval $[\plo(x),\pup(x)]$. Since $\dcsm(\Gamma)$ constrains $a$ only
\emph{pointwise} and $a$ is otherwise an arbitrary measurable function, the set of
achievable regressions is the product $\prod_x[\plo(x),\pup(x)]$, i.e.\ the
stated box. It is non-empty ($a\equiv1$ gives $p=p_1$), convex and compact
(a product of compact intervals, in the $L^\infty$ or weak topology as
appropriate). Nestedness and the two limits are immediate from monotonicity of
$\sigma$ and $\sigma(\pm\infty)=\{0,1\}$.
\end{proof}

\subsection{Ranking metrics}
\label{app:ranking}

\begin{proof}[Proof of \cref{thm:rank-identity}]
Let $(X,Y)$, $(X',Y')$ be independent with $X,X'\sim\mu$ and
$\Prb(Y=1\mid X)=p(X)$. From \eqref{eq:auc-def} the numerator is
\[
  N(p)=\iint \psi\big(f(x),f(x')\big)\,p(x)\,\big(1-p(x')\big)\,d\mu(x)\,d\mu(x').
\]
Define the operator $(\mathcal{K}h)(x)=\int\psi(f(x),f(x'))h(x')\,d\mu(x')$ on
$L^2(\mu)$, so $N(p)=\langle p, \mathcal{K}(\one-p)\rangle
= \langle p, \mathcal{K}\one\rangle - \langle p,\mathcal{K}p\rangle$ and
$\mathcal{K}\one = R$ by \eqref{eq:midrank}. Since
$\psi(u,v)+\psi(v,u)=1$ for all $u,v$ (including $u=v$, where both sides are
$\tfrac12+\tfrac12$), for any $g,h$
\[
  \langle g,\mathcal{K}h\rangle + \langle h,\mathcal{K}g\rangle
  = \iint\big[\psi(f(x),f(x'))+\psi(f(x'),f(x))\big]g(x)h(x')\,d\mu\,d\mu
  = \Big(\int g\,d\mu\Big)\Big(\int h\,d\mu\Big),
\]
i.e.\ $\mathcal{K}+\mathcal{K}^{*}=\one\otimes\one$. Taking $g=h=p$ gives
$2\langle p,\mathcal{K}p\rangle = \pi^{2}$, so
$\langle p,\mathcal{K}p\rangle = \pi^{2}/2$ for \emph{every} $p$. Hence
$N(p) = \langle p,R\rangle - \pi^2/2$, and dividing by
$\Prb(Y=1)\Prb(Y'=0)=\pi(1-\pi)$ gives \eqref{eq:rank-identity}.
\end{proof}

\begin{proof}[Proof of \cref{thm:sharp-auc}]
\emph{Reduction.} By \cref{lem:box} the identified set is the box $\I_\Gamma$.
Write $\Pi(p)=\E_\mu[p]$. For fixed $\pi\in(0,1)$ the denominator of
\eqref{eq:rank-identity} is a positive constant, so on the slice
$\{p\in\I_\Gamma : \Pi(p)=\pi\}$ the map $p\mapsto\AUC(p)$ is an increasing
affine function of $\E_\mu[pR]$. Therefore
\[
  \sup_{p\in\I_\Gamma}\AUC(p)
  = \sup_{\pi\in[\pi_-,\pi_+]}\ \sup_{\substack{p\in\I_\Gamma\\ \Pi(p)=\pi}}\AUC(p)
  = \sup_{\pi\in[\pi_-,\pi_+]}\frac{V^{+}(\pi)-\pi^2/2}{\pi(1-\pi)},
\]
and symmetrically for the infimum. The outer range is exactly $[\pi_-,\pi_+]$:
$\Pi$ is continuous on the connected set $\I_\Gamma$ with
$\Pi(\plo)=\pi_-$, $\Pi(\pup)=\pi_+$, so by the intermediate value theorem every
intermediate $\pi$ is attained, and no other value is.

\emph{(i) Structure of $V^{\pm}$.} Substituting $p=\plo + u$ with
$0\le u(x)\le \delta(x)$ turns the inner problem into a continuous
(fractional) knapsack: maximise $\E_\mu[uR]$ subject to $\E_\mu[u]=\pi-\pi_-$ and
$0\le u\le\delta$. The greedy solution --- saturate $u=\delta$ in decreasing
order of $R$ until the budget is spent, with at most one fractional unit ---
is optimal by the standard exchange argument. Its value as a function of the
budget is piecewise linear with slopes equal to the sorted $R$ values, which are
decreasing; hence $V^{+}$ is concave with at most $n$ breakpoints. For
$V^{-}$ the order is reversed and the slopes increase, giving convexity.

\emph{(ii) Stationary points.} On a linear piece write
$V^{\pm}(\pi)=\alpha+\beta\pi$ and $h(\pi)=N(\pi)/D(\pi)$ with
$N=\alpha+\beta\pi-\pi^2/2$ and $D=\pi-\pi^2$. Then
\[
  N'D-ND'
  = (\beta-\pi)(\pi-\pi^2) - \big(\alpha+\beta\pi-\tfrac{\pi^2}{2}\big)(1-2\pi)
  = \big(\beta-\tfrac12\big)\pi^{2} + 2\alpha\pi - \alpha,
\]
the $\pi^3$ terms cancelling identically. Setting this to zero gives the stated
quadratic. (The general form in \cref{prop:unified} follows by the same
expansion with $N=n_0+n_1\pi+n_2\pi^2$, $D=d_0+d_1\pi+d_2\pi^2$, yielding
$N'D-ND' = (n_1d_0-n_0d_1) + 2(n_2d_0-n_0d_2)\pi + (n_2d_1-n_1d_2)\pi^2$.)

\emph{(iii) Complexity.} Computing $R$ is a sort, $O(n\log n)$. Sorting by $R$
and forming the cumulative sums gives both profiles in $O(n\log n)$. Each of the
$\le n$ pieces contributes $O(1)$ candidate points (two endpoints, at most two
roots), each evaluated in $O(1)$, so the outer scan is $O(n)$.

\emph{Attainment.} The optimising $\pi^\star$ is attained (the objective is
continuous on the compact $[\pi_-,\pi_+]\cap(0,1)$ after the usual convention at
the degenerate ends), and the greedy construction returns an explicit
$p^\star\in\I_\Gamma$ with $\Pi(p^\star)=\pi^\star$ and
$\E_\mu[p^\star R]=V^{\pm}(\pi^\star)$. By \cref{lem:box} that $p^\star$
corresponds to an admissible tilt and hence to an admissible law.
\end{proof}

\begin{proof}[Proof of \cref{prop:corner}]
$\plo$, $\pup$ and $\tilde p$ all belong to $\I_\Gamma$, so their AUROC values
lie in the sharp interval and $\mathcal{C}$ is contained in it. For strictness,
consider the upper end. By the greedy characterisation, the maximiser at the
optimal prevalence sets $p=\pup$ on $\{R > r^\star\}$ and $p=\plo$ on
$\{R<r^\star\}$ for a pivot $r^\star$. If $R$ is non-constant on
$\{\delta>0\}$ this differs from $\plo$, $\pup$ and $\tilde p$ on a set of
positive measure; since $\E_\mu[pR]$ is strictly increasing in $p$ where $R>0$
and the objective is strictly increasing in $\E_\mu[pR]$ at fixed prevalence,
the maximum strictly exceeds all three corner values unless the optimal
prevalence coincides with one corner's prevalence \emph{and} the corner already
solves the knapsack, which requires $R$ constant on $\{\delta>0\}$. An explicit
three-point counterexample is given in \cref{app:counterexample}.
\end{proof}

\begin{proof}[Proof of \cref{thm:dcl}]
\emph{(a)} By \cref{lem:box}, $\I_\Gamma$ is a product of intervals and
$R_P(f)=\E_\mu[p\,\ell_1+(1-p)\ell_0]$ is affine in $p(x)$ pointwise with slope
$\ell_1-\ell_0$. The supremum therefore decouples and is attained at
$\pup(x)$ where $\ell_1\ge\ell_0$ and at $\plo(x)$ otherwise:
\[
  \sup_{p\in\I_\Gamma}R_P(f)
  = \E_\mu\Big[\max\big\{\pup\ell_1+(1-\pup)\ell_0,\ \plo\ell_1+(1-\plo)\ell_0\big\}\Big].
\]
Apply $\max(A,B)=\tfrac{A+B}{2}+\tfrac{|A-B|}{2}$ with
$A-B=(\pup-\plo)(\ell_1-\ell_0)=\delta\,(\ell_1-\ell_0)$ to get
\eqref{eq:wc-decomp}. For the logistic loss
$\ell(z,1)-\ell(z,0)=\log(1+e^{-z})-\log(1+e^{z})=-z$, so $g_\ell(z)=|z|$.

\emph{(b)} For each fixed $p\in[0,1]$ the map
$z\mapsto p\,\ell(z,1)+(1-p)\ell(z,0)$ is a non-negative combination of convex
functions, hence convex. A pointwise supremum of convex functions is convex, so
$z\mapsto\max_{p\in[\plo,\pup]}\{\cdot\}$ is convex, and
$\Rbar_\Gamma(f)=\E_\mu[\cdot]$ is convex in $f$ as an element of any convex
class. For the second claim, subtracting the (convex) midpoint term from a
convex function leaves $\tfrac12\delta\,g_\ell$, which is convex iff $g_\ell$ is,
whenever $\delta$ may be taken constant and positive on a set of positive
measure. For the hinge loss $g_\ell(z)=2|z|$ on $|z|\le1$ and $1+|z|$ outside,
whose one-sided slopes at $z=1$ are $2$ then $1$: not convex.

\emph{(c)} $R_P(f)$ is convex in $f$ over the convex set $\F$ and affine --- hence
concave and continuous --- in $p$ over the convex compact $\I_\Gamma$. Sion's
minimax theorem \citep{sion1958general} gives $\inf\sup=\sup\inf$ and existence
of a saddle point; the adversary's best response is the vertex rule from part
(a). At a saddle point $\hat f$ minimises $R_{P^\star}(\cdot)$, i.e.\ it is the
Bayes predictor for $P^\star$.

\emph{(d)} Minimise $J(z)=\tilde p\,\zeta(-z)+(1-\tilde p)\zeta(z)
+\tfrac{\delta}{2}|z|$ pointwise, $\zeta(u)=\log(1+e^{u})$. For $z\neq0$,
$J'(z)=\sigma(z)-\tilde p+\tfrac{\delta}{2}\operatorname{sign}(z)$. On $z>0$,
$J'=0$ gives $\sigma(z)=\tilde p-\tfrac{\delta}{2}=\plo$, so
$z=\logit\plo$, admissible iff $\plo>\tfrac12$. On $z<0$, $J'=0$ gives
$\sigma(z)=\tilde p+\tfrac{\delta}{2}=\pup$, so $z=\logit\pup$, admissible iff
$\pup<\tfrac12$. Otherwise $0$ is a minimiser because the subdifferential at $0$
is $[\tfrac12-\pup,\ \tfrac12-\plo]$, which contains $0$ exactly when
$\plo\le\tfrac12\le\pup$. The three cases are exhaustive and mutually exclusive,
and $J$ is strictly convex, so the minimiser is unique.
\end{proof}

\begin{proof}[Proof of \cref{prop:regret}]
For the logistic loss and $q=\sigma(z)$,
$\E[\ell(z,Y)\mid p] - \inf_{z'}\E[\ell(z',Y)\mid p] = \KL(p\|q)$, so the
pointwise problem is $\min_{q}\max_{p\in[\plo,\pup]}\KL(p\|q)$. For fixed $q$,
$p\mapsto\KL(p\|q)$ is convex, so its maximum over an interval is attained at an
endpoint; the objective is thus
$\Psi(q)=\max\{\KL(\plo\|q),\KL(\pup\|q)\}$. Each branch is strictly convex in
$q$ with minimum $0$ at $q=\plo$ and $q=\pup$ respectively, the first increasing
in $q$ on $q>\plo$ and the second decreasing on $q<\pup$; hence on $(\plo,\pup)$
the two branches cross exactly once and $\Psi$ is minimised where they are equal
(for $\plo<\pup$). Setting $\KL(\plo\|q)=\KL(\pup\|q)$ and writing
$H(u)=-u\log u-(1-u)\log(1-u)$,
\[
  -H(\plo)-\plo\log q-(1-\plo)\log(1-q)
  \;=\;
  -H(\pup)-\pup\log q-(1-\pup)\log(1-q),
\]
so $H(\pup)-H(\plo) = (\plo-\pup)\log q + (\pup-\plo)\log(1-q)
= -(\pup-\plo)\logit q$, giving \eqref{eq:regret-rule}. As $\pup\downarrow\plo=p$
the right-hand side tends to $-H'(p)=\logit p$, recovering the precise case.
\end{proof}

\begin{proof}[Proof of \cref{prop:midrank}]
Fix an ordering $\sigma$ and let $R_\sigma(i)\in\{\tfrac{1}{2n},\tfrac{3}{2n},\dots\}$
be the mid-rank of unit $i$ under $\sigma$ (no ties, so the multiset of ranks
is the same for every $\sigma$). Write $p=\plo+\delta u$ with $u\in[0,1]^n$ and
$\pi=\bar p=\bar{\plo}+\delta\bar u$. By \cref{thm:rank-identity}, at fixed
$\pi$ the AUROC is $(\langle p,R_\sigma\rangle_n-\pi^2/2)/(\pi(1-\pi))$ with
$\langle p,R_\sigma\rangle_n=\langle\plo,R_\sigma\rangle_n+\delta\langle u,R_\sigma\rangle_n$.
For fixed $\bar u=m$,
$\min\{\langle u,R_\sigma\rangle_n:u\in[0,1]^n,\bar u=m\}$ is attained by
filling the $nm$ smallest ranks (continuous knapsack), so it equals a function
$\Phi^-(m)$ of the multiset of ranks alone --- the same for every $\sigma$; likewise
$\max\{\cdot\}=\Phi^+(m)$. Hence
$V^-_\sigma(\pi)=\langle\plo,R_\sigma\rangle_n+\delta\Phi^-(m)$ and
$V^+_\sigma(\pi)=\langle\plo,R_\sigma\rangle_n+\delta\Phi^+(m)$. By the
rearrangement inequality, $\langle\plo,R_\sigma\rangle_n$ is maximised exactly
when $\sigma$ orders the units by $\plo$ (ties in $\plo$ may be broken
arbitrarily), call it $\sigma^\star$. Thus $V^\pm_{\sigma^\star}(\pi)\ge
V^\pm_\sigma(\pi)$ for every admissible $\pi$, the denominator $\pi(1-\pi)$
does not depend on $\sigma$, and taking $\min_\pi$ (resp.\ $\max_\pi$) of
the ratio preserves the inequality:
$\min_p\AUC(\sigma^\star,p)\ge\min_p\AUC(\sigma,p)$ and
$\max_p\AUC(\sigma^\star,p)\ge\max_p\AUC(\sigma,p)$. Since
$\tilde p=\plo+\delta/2$ and $\pup=\plo+\delta$ induce the same ordering, the
three plug-in orderings coincide. For unequal widths the knapsack has
capacities $\delta_i$, its value depends on which capacities occupy the low
ranks, and the decoupling fails; \cref{app:audit-ranking} exhibits explicit
instances.
\end{proof}

\begin{proof}[Proof of \cref{cor:validity}]
Immediate: $P\in\I_\Gamma$ and $\Rbar_\Gamma(f)$ is the supremum of $R_{P'}(f)$
over $P'\in\I_\Gamma$.
\end{proof}

\subsection{Impossibility}
\label{app:impossibility}

\begin{proof}[Proof of \cref{thm:impossible}]
It suffices to exhibit two laws in $\mathcal{P}_{\Gamma,\eta}$ with identical
observed-data distributions and deployment risks differing by $\Delta$.

Let $P_X$ be arbitrary, let $e(x)=1-\eta$ so $\Prb(T=0)=\eta$, and fix any
$p_1(\cdot)$. Define two laws $P_{\pm}$ that agree on $(P_X, e, p_1)$ --- hence
on $P_{\mathrm{obs}}$, since the observed-data law is a function of exactly
$(P_X,e,p_1)$ and nothing else --- and differ in the censored branch:
\[
  \odds\big(p_0^{\pm}(x)\big) \;=\; \Gamma^{\pm1}\,\odds\big(p_1(x)\big).
\]
Both satisfy $\dcsm(\Gamma)$ with equality, so both lie in
$\mathcal{P}_{\Gamma,\eta}$. Their deployment risks satisfy
\[
  R_{P_+}(f)-R_{P_-}(f)
  = \E\Big[(1-e(X))\big(p_0^{+}(X)-p_0^{-}(X)\big)
    \big(\ell(f(X),1)-\ell(f(X),0)\big)\Big],
\]
whose absolute value is $\Delta$ as stated after noting that for
$p_1=\tfrac12$, $p_0^{+}-p_0^{-}=\tfrac{\Gamma}{\Gamma+1}-\tfrac{1}{\Gamma+1}
=\tfrac{\Gamma-1}{\Gamma+1}$ (for general $p_1$, replace this factor by the
corresponding difference, which is what the conditional expectation in
\eqref{eq:impossible} records).

Now let $\hat R_n$ be any estimator and set
$A=\{|\hat R_n - R_{P_-}(f)| < \Delta/2\}$. On $A^{c}$ the estimator errs by at
least $\Delta/2$ under $P_-$; on $A$, the triangle inequality gives
$|\hat R_n - R_{P_+}(f)| \ge \Delta - \Delta/2 = \Delta/2$, so it errs by at
least $\Delta/2$ under $P_+$. Because $P_{\mathrm{obs}}$ is the same under both
laws and $\hat R_n$ is a measurable function of the observed sample,
$\Prb_{P_-}(A)=\Prb_{P_+}(A)$. Hence
\[
  \max_{\pm}\Prb_{P_\pm}\!\big(|\hat R_n - R_{P_\pm}(f)|\ge\Delta/2\big)
  \;\ge\;
  \max\big\{1-\Prb_{P_-}(A),\ \Prb_{P_-}(A)\big\}
  \;\ge\;\tfrac12 .
\]
The argument uses no asymptotics, so the conclusion holds at every $n$.
\end{proof}

\begin{proof}[Proof of \cref{cor:gamma-unidentified}]
The observed-data law is determined by $(P_X, e, p_1)$, and $\dcsm(\Gamma)$
constrains only $p_0$, which does not enter it. So for every $\Gamma\ge1$ and
every observed-data law there exists a compatible $P$ (take $a\equiv1$). The
compatible sets coincide, and no test can have power.
\end{proof}

\begin{proof}[Proof of \cref{prop:falsify}]
Under \cref{ass:iv}, $Z\perp(Y,S)\mid X$ implies $\Prb(Y=1\mid X=x,Z=z)=p(x)$
for every $z$; applying \cref{lem:box} within each level $z$ gives
$p(x)\in[\plo_z^\Gamma(x),\pup_z^\Gamma(x)]$, hence membership in the
intersection. Each interval is continuous and nested increasing in $\Gamma$
(monotonicity of $\sigma$), so $\Gamma\mapsto\one\{\bigcap_z\neq\emptyset\}$ is
non-decreasing and the infimum defining $\Gamma_{\min}$ is attained and found by
bisection. Since the true $\Gamma_0$ makes the intersection non-empty (it
contains $p(x)$), $\Gamma_{\min}(x)\le\Gamma_0$. Finally, if
$\Gamma<\Gamma_{\min}(x)$ the intersection is empty, so no law satisfying
$\dcsm(\Gamma)$ conditionally on each $z$ can have generated the data, i.e.\
$\dcsm(\Gamma)$ is refuted.
\end{proof}

\subsection{Generalisation}
\label{app:generalization}

\begin{proof}[Proof of \cref{thm:generalization}]
Write $\bar\ell(z;x)=\max_{p\in[\plo(x),\pup(x)]}\{p\ell(z,1)+(1-p)\ell(z,0)\}$.

\emph{Lipschitzness.} For fixed $x$ the maximising $p$ is a vertex, and
$\partial_z[p\ell(z,1)+(1-p)\ell(z,0)]$ is a convex combination of
$\partial_z\ell(z,1)$ and $\partial_z\ell(z,0)$, each bounded by $L$; by Danskin's
theorem $\bar\ell(\cdot;x)$ is $L$-Lipschitz. For the logistic loss
$\partial_z\ell(z,1)=-\sigma(-z)\in[-1,0]$ and
$\partial_z\ell(z,0)=\sigma(z)\in[0,1]$, so $L=1$.

\emph{Uniform deviation.} $\bar\ell$ takes values in $[0,\ell_{\max}(B)]$ on
$\|f\|_\infty\le B$. By symmetrisation and Talagrand's contraction inequality
\citep{bartlett2002rademacher}, with probability $\ge1-\delta$,
\[
  \sup_{f\in\F}\big|\hat{\Rbar}^{\,\mathrm{oracle}}_\Gamma(f)-\Rbar_\Gamma(f)\big|
  \;\le\; 2L\,\Rad_n(\F)
  \;+\;\ell_{\max}(B)\sqrt{\tfrac{\log(2/\delta)}{2n}},
\]
where $\hat{\Rbar}^{\,\mathrm{oracle}}$ uses the true $(\plo,\pup)$. The standard
argument $\Rbar_\Gamma(\hat f) - \inf_\F\Rbar_\Gamma
\le 2\sup_\F|\hat{\Rbar}-\Rbar|$ doubles both terms.

\emph{Nuisance error.} $\bar\ell$ depends on $(\plo,\pup)$ through the affine
coefficient $\ell_1-\ell_0$, so
$|\bar\ell(z;\hat{\plo},\hat{\pup})-\bar\ell(z;\plo,\pup)|
\le \max(|\hat{\plo}-\plo|,|\hat{\pup}-\pup|)\cdot g_\ell(z)
\le \kappa(B)\max(|\hat{\plo}-\plo|,|\hat{\pup}-\pup|)$.
Both bounds are of the form $e\,p_1+(1-e)\sigma(\eta_1\mp\log\Gamma)$; the map
$(e,\eta_1)\mapsto\plo$ has partial derivatives bounded by
$|p_1-\sigma(\eta_1-\log\Gamma)|\le1$ in $e$ and by
$\sup|\sigma'|=\tfrac14$ in $\eta_1$, so
$|\hat{\plo}-\plo|\le|\hat e-e|+\tfrac14|\hat\eta_1-\eta_1|$, and likewise for
$\pup$. Taking expectations and doubling (again for the excess-risk argument)
gives the third term of \eqref{eq:gen-bound}. Note the bound is on the
\emph{logit} scale in $p_1$: this is where the $1$-Lipschitz behaviour of
$\Gamma$ as a translation is used.
\end{proof}

\begin{proof}[Proof of \cref{cor:loggamma}]
By \eqref{eq:bayes-act} the population optimum has
$|f^\star_\Gamma(x)|\le\max\{|\logit\plo(x)|,|\logit\pup(x)|\}$. Since
$\plo\ge e p_1 + (1-e)\sigma(\eta_1-\log\Gamma)$ and
$\pup\le e p_1 + (1-e)\sigma(\eta_1+\log\Gamma)$, monotonicity of $\logit$ and
$\logit\sigma(u)=u$ give $|\logit\pup|\le|\eta_1|+\log\Gamma\le B_0+\log\Gamma$
and symmetrically for $\plo$. Substituting $B=B_0+\log\Gamma$ and
$\ell_{\max}(B)\le\log2+B$, $\kappa(B)=B$ for the logistic loss into
\eqref{eq:gen-bound} gives \eqref{eq:loggamma}.
\end{proof}

\begin{proof}[Sketch of \cref{prop:lower-gen}]
Take $\X$ a single point, so $\F_\Gamma$ is the set of constants
$|z|\le B_0+\log\Gamma$, and let the identified set be
$[\plo,\pup]$ with $\pup<\tfrac12$ and $\logit\pup\asymp-(B_0+\log\Gamma)$. The
empirical objective is an average of $n$ i.i.d.\ terms whose range is
$\Theta(B_0+\log\Gamma)$ and whose minimiser must be located to accuracy
$\varepsilon$ in a loss that is locally quadratic with curvature bounded below;
a two-point argument on the Bernoulli parameter (Le Cam) then shows that
distinguishing the two candidate optima requires
$n = \Omega\big((\log\Gamma)^2/\varepsilon^2\big)$, i.e.\
$\varepsilon = \Omega(\log\Gamma\cdot n^{-1/2})$.
\end{proof}

\subsection{Nuisance bottleneck}
\label{app:nuisance}

Throughout, $g_\gamma(p)=\gamma p/(1+(\gamma-1)p)$, so
$g_\gamma'(p)=\gamma/(1+(\gamma-1)p)^2$ and
$g_\gamma''(p)=-2\gamma(\gamma-1)/(1+(\gamma-1)p)^3$. For
$\gamma\in\{\Gamma,1/\Gamma\}$ and $p\in[0,1]$ one checks
$0<g_\gamma'\le\Gamma$ and $|g_\gamma''|\le2\Gamma(\Gamma-1)$: for
$\gamma=\Gamma$ both suprema are at $p=0$; for $\gamma=1/\Gamma$ both are at
$p=1$, where $1+(\gamma-1)p=1/\Gamma$.

\begin{proof}[Proof of \cref{thm:nuisance}(a)]
Since $\E[TY\mid X]=e\,p_1$ and $g_\gamma(p_1(X))$ is a function of $X$,
$\pi_\gamma=\E[TY]+\E[(1-e)\,g_\gamma(p_1)]$; the first term is a sample mean
and contributes no nuisance bias, so consider
$\psi=\E[(1-T)g_\gamma(p_1(X))]$. Conditioning on $X$ in \eqref{eq:onestep},
\[
  \E\big[(1-T)g_\gamma(\hat p_1)+(1-\hat e)\,g_\gamma'(\hat p_1)\,
  \tfrac{T(Y-\hat p_1)}{\hat e}\big]
  = \E\big[(1-e)\,g_\gamma(\hat p_1)\big]
  + \E\big[(1-\hat e)\tfrac{e}{\hat e}\,g_\gamma'(\hat p_1)\,(p_1-\hat p_1)\big].
\]
Subtract $\psi=\E[(1-e)g_\gamma(p_1)]$ and expand $g_\gamma(p_1)$ around
$\hat p_1$ to second order,
$g_\gamma(p_1)=g_\gamma(\hat p_1)+g_\gamma'(\hat p_1)(p_1-\hat p_1)
+\tfrac12 g_\gamma''(\xi)(p_1-\hat p_1)^2$:
\[
  \E[\hat\psi]-\psi
  = \E\Big[g_\gamma'(\hat p_1)\,(p_1-\hat p_1)\Big\{(1-\hat e)\tfrac{e}{\hat e}-(1-e)\Big\}\Big]
  - \tfrac12\E\big[(1-e)\,g_\gamma''(\xi)\,(p_1-\hat p_1)^2\big].
\]
The brace equals $(e-\hat e)/\hat e$. Bounding $g_\gamma'\le\Gamma$,
$\hat e\ge\varepsilon$, $1-e\le1$, $|g_\gamma''|\le2\Gamma(\Gamma-1)$ and
applying Cauchy--Schwarz to the first term gives \eqref{eq:dr-bias}. For the
plug-in, the same expansion without the correction term leaves
$\E[(1-e)g_\gamma'(\hat p_1)(\hat p_1-p_1)]+O(\|\hat p_1-p_1\|_2^2)$, whose
leading term is of first order in $\|\hat p_1-p_1\|_1$ and does not vanish
unless $\hat p_1$ is unbiased for $p_1$ pointwise.

Asymptotics follow the standard cross-fitting argument
\citep{chernozhukov2018double}: with nuisances fit on independent folds, the
estimator is $\Prb_n[\varphi_\gamma(\hat e,\hat p_1)]$ with
$\varphi_\gamma(e,p_1)=TY+(1-T)g_\gamma(p_1)+(1-e)g_\gamma'(p_1)T(Y-p_1)/e$
the (uncentred) efficient influence function, the empirical-process term is
$o_P(n^{-1/2})$ by fold independence and $L_2$-consistency of the nuisances,
and the remaining bias is \eqref{eq:dr-bias}, assumed $o_P(n^{-1/2})$. The
central limit theorem and Slutsky give the normal limit with the plug-in
standard error $\hat s_\gamma^2=\Prb_n[(\varphi_\gamma-\hat\pi_\gamma)^2]/n$.
For the outer interval, the union bound gives
$\Prb\{\pi_{1/\Gamma}<L\ \text{or}\ \pi_\Gamma>U\}\le\alpha/2+\alpha/2$,
each one-sided miss having asymptotic probability $\alpha/2$ at
$z_{1-\alpha/2}$.
\end{proof}

\begin{proof}[Proof of \cref{thm:nuisance}(b)]
With $\eta=\logit p_1$ and $s_\pm=\sigma(\eta\pm\log\Gamma)$,
$\plo=e\,\sigma(\eta)+(1-e)s_-$ and $\pup=e\,\sigma(\eta)+(1-e)s_+$. Then
$\partial_e\plo=\sigma(\eta)-s_-\ge0$, $\partial_e\pup=\sigma(\eta)-s_+\le0$,
and $\partial_\eta\plo=e\,\sigma'(\eta)+(1-e)\sigma'(\eta-\log\Gamma)>0$,
likewise for $\pup$; the four monotonicities follow. If $e\in[e_L,e_U]$ and
$\eta\in[\eta_L,\eta_U]$ then
$\plo(e,\eta)\ge\plo(e_L,\eta)\ge\plo(e_L,\eta_L)$ and
$\pup(e,\eta)\le\pup(e_L,\eta)\le\pup(e_L,\eta_U)$, which is
\eqref{eq:inflated}. The coverage inequality is the inclusion of events
$\{e\in[e_L,e_U],\eta\in[\eta_L,\eta_U]\}\subseteq\{p\in\I^\circ\}$, using
$p\in[\plo,\pup]$ (\cref{lem:box}). A functional $F$ is monotone in the box
if $I\subseteq I'$ pointwise implies $\inf_{p\in I'}F\le\inf_{p\in I}F$ and
$\sup_{p\in I}F\le\sup_{p\in I'}F$; the sharp endpoints
\eqref{eq:sharp-auc}, $\Rbar_\Gamma$ \eqref{eq:wc-decomp} and the regret
value of \cref{prop:regret} are all of this form, so their values on $\I^\circ$
bracket their values on the true box.
\end{proof}

\begin{proof}[Proof of \cref{thm:nuisance}(c)]
Given the bins (fixed by the estimation split and independent of the
calibration split), the calibration counts $\sum_{i\in B_k}T_i$ and
$\sum_{i\in B_k}T_iY_i$ are binomial with success probabilities $e^k$ and
$p_1^k$ conditionally on the bin sizes, so the Clopper--Pearson intervals at
level $1-\alpha/2$ each cover with probability at least $1-\alpha/2$, and the
union bound gives $1-\alpha$ jointly. If $e$ and $p_1$ are constant on $B_k$
then $e(x)=e^k$ and $p_1(x)=p_1^k$ for $x\in B_k$ and coverage is pointwise;
otherwise $|e(x)-e^k|\le\omega_k$ for every $x\in B_k$ (a bin mean lies within
the bin's oscillation of every bin value), so widening by $\omega_k$ restores
pointwise coverage, and similarly for $p_1$ on the logit scale with the
oscillation of $\eta_1$. The implementation replaces the unknown $\omega_k$
by the within-bin range of the \emph{estimate}, which is exact when the
estimate is a monotone recalibration of the truth on the bin and otherwise an
approximation, stated as such in \cref{tab:nuisance}. That no assumption-free
pointwise interval exists is \citet{low1997nonparametric}: any confidence
interval for $f(x_0)$ that is honest over a class containing functions with
arbitrarily sharp local features has expected length bounded away from zero.
\end{proof}

\subsection{Uncertainty}
\label{app:uq}

\paragraph{Notation.} $H(q)=-q\log q-(1-q)\log(1-q)$ (nats);
$\I_\Gamma(x)=[\plo(x),\pup(x)]$; $\delta(x)=\pup(x)-\plo(x)$; $\eta_1=\logit p_1$.
For a closed $C\subseteq[0,1]$ the \emph{credal epistemic term} is
$\mathrm{EU}(C)=\max_{q\in C}H(q)-\min_{q\in C}H(q)$ \citep{abellan2005difference};
$\mathrm{Cens}(x)=\mathrm{EU}(\I_\Gamma(x))$ is the censoring term \eqref{eq:cens}; and
$m(x)=\min_{q\in\I_\Gamma(x)}|\logit q|$, which is $0$ iff $\tfrac12\in\I_\Gamma(x)$.

\begin{definition}[Valid credal set, reported and honest epistemic terms]
\label{def:valid}
Fix $x$. A closed set $C\subseteq[0,1]$ is \emph{valid at $x$ under $\dcsm(\Gamma)$} if
$p(x)\in C$ for every law $P$ that satisfies $\dcsm(\Gamma)$ and induces the observed law
$P_{\mathrm{obs}}$. An \emph{observed-data method} returns, for each $n$, a credal set
$C_n(x)\subseteq[0,1]$ built from $O_1,\dots,O_n$ --- the interval hull of an ensemble
$\{p^{(n)}_m(x)\}_{m\le M}$ fit to the selected units, or the support of a posterior over
$p_\theta(x)$ --- and reports the two-way decomposition
\[
  \mathrm{Tot}_n(x)=H(\bar p_n(x)),\qquad
  \mathrm{Ale}_n(x)=\tfrac1M\textstyle\sum_m H(p^{(n)}_m(x)),\qquad
  \mathrm{Epi}_n(x)=\mathrm{Tot}_n(x)-\mathrm{Ale}_n(x),
\]
with $\bar p_n=\tfrac1M\sum_m p^{(n)}_m$ (posterior expectations in the Bayesian case).
The \emph{honest epistemic term} of the method is the credal epistemic term of the smallest
set that is both consistent with its own sampling spread and valid,
$\mathrm{EU}\big(\mathrm{hull}(C_n(x)\cup\I_\Gamma(x))\big)$, and its \emph{bias} is
\begin{equation}
  \mathrm{Bias}_n(x)\;:=\;\mathrm{EU}\big(\mathrm{hull}(C_n(x)\cup\I_\Gamma(x))\big)\;-\;\mathrm{Epi}_n(x).
  \label{eq:uq-bias}
\end{equation}
\end{definition}

\begin{corollary}[Full statement of \cref{cor:uq}]
\label{cor:uq-full}
Assume $\dcsm(\Gamma)$ and fix $x$ with $e(x)<1$, $p_1(x)\in(0,1)$ and $\Gamma>1$
(otherwise $\delta(x)=0$ and every quantity below is $0$). Then:
\begin{enumerate}
\item[(i)] \textup{(The naive answer is never refuted.)}
  $\plo(x)<p_1(x)<\pup(x)$. The selection-on-observables law $P^{(1)}$
  (tilt $a\equiv1$, so $p=p_1$) satisfies $\dcsm(\Gamma)$ for every $\Gamma\ge1$ and induces
  the same $P_{\mathrm{obs}}$; consequently every statistic $S(O_1,\dots,O_n)$ has the same
  distribution under every admissible law. In particular no calibration, coverage or
  goodness-of-fit test computed on observed data has power above its size against
  ``$\mathrm{Epi}_n$ understates the epistemic uncertainty about $p(x)$''.
\item[(ii)] \textup{(The honest floor and a finite-$n$ inequality.)}
  $C$ is valid at $x$ iff $C\supseteq\I_\Gamma(x)$, and
  \[
    \inf\{\mathrm{EU}(C):\ C\ \text{valid at }x\}\;=\;\mathrm{EU}(\I_\Gamma(x))\;=\;\mathrm{Cens}(x),
  \]
  attained by $C=\I_\Gamma(x)$. For every $n$ and every observed-data method,
  $0\le\mathrm{Epi}_n(x)\le\mathrm{EU}(C_n(x))$ and
  \begin{equation}
    \mathrm{Bias}_n(x)\;\ge\;\max\big\{\mathrm{EU}(C_n(x)),\ \mathrm{Cens}(x)\big\}-\mathrm{Epi}_n(x)
    \;\ge\;\mathrm{Cens}(x)-\mathrm{Epi}_n(x).
    \label{eq:uq-bias-finite}
  \end{equation}
\item[(iii)] \textup{(Asymptotic overconfidence.)} Let
  $\varepsilon_n(x)=\sup_{q\in C_n(x)}|q-p_1(x)|$ and suppose $\varepsilon_n(x)\to0$
  (almost surely, or in probability; the conclusions then hold in the same mode) --- as for a
  consistent posterior or a converged ensemble of a well-specified class. Then
  \[
  \begin{gathered}
    \mathrm{Epi}_n(x)\;\le\;H\big(\min\{2\varepsilon_n(x),\tfrac12\}\big)\;\longrightarrow\;0,\\
    \mathrm{Cens}(x)-\mathrm{Epi}_n(x)\;\le\;\mathrm{Bias}_n(x)\;\le\;
    \mathrm{Cens}(x)+2H\big(\min\{\varepsilon_n(x),\tfrac12\}\big)-\mathrm{Epi}_n(x),
  \end{gathered}
  \]
  so $\mathrm{Bias}_n(x)\to\mathrm{Cens}(x)>0$. Precisely: \emph{``asymptotically
  overconfident by at least $\mathrm{Cens}(x)$'' means}
  \begin{equation}
    \liminf_{n\to\infty}\ \inf_{C\ \mathrm{valid\ at}\ x}\big[\mathrm{EU}(C)-\mathrm{Epi}_n(x)\big]
    \;=\;\lim_{n\to\infty}\big[\mathrm{Cens}(x)-\mathrm{Epi}_n(x)\big]\;=\;\mathrm{Cens}(x),
    \label{eq:uq-overconf}
  \end{equation}
  i.e.\ for every $\epsilon>0$ there is $n_0$ such that for all $n\ge n_0$ the reported
  epistemic term is at least $\mathrm{Cens}(x)-\epsilon$ below the epistemic term of
  \emph{every} valid credal set, while $\mathrm{Cens}(x)$ itself does not depend on $n$.
\item[(iv)] \textup{(Explicit form of the censoring term.)} With
  $d(x)=\max\{\tfrac12-\plo(x),\ \pup(x)-\tfrac12\}$,
  \begin{align}
    \tfrac12\notin\I_\Gamma(x):&\quad
      \mathrm{Cens}(x)=\Big|\int_{\plo(x)}^{\pup(x)}\logit q\,dq\Big|
      \;\ge\;\delta(x)\,m(x)+2\,\delta(x)^2,\qquad
      \mathrm{Cens}(x)\le\delta(x)\max_{q\in\I_\Gamma(x)}|\logit q|;
      \label{eq:cens-mono}\\
    \tfrac12\in\I_\Gamma(x):&\quad
      \mathrm{Cens}(x)=\log2-\min\{H(\plo),H(\pup)\}
      =\KL\big(\mathrm{Bern}(q^\ast)\,\|\,\mathrm{Bern}(\tfrac12)\big)
      \;\ge\;2\,d(x)^2\;\ge\;\tfrac12\,\delta(x)^2,
      \label{eq:cens-inside}
  \end{align}
  where $q^\ast$ is the endpoint of $\I_\Gamma(x)$ farther from $\tfrac12$. In both cases
  \begin{equation}
    \mathrm{Cens}(x)\;\ge\;\delta(x)\,m(x)+\tfrac12\,\delta(x)^2,
    \qquad
    \delta(x)=\big(1-e(x)\big)\,\frac{\odds(p_1(x))\,(\Gamma^2-1)}
        {\big(1+\Gamma\,\odds(p_1(x))\big)\big(\Gamma+\odds(p_1(x))\big)}\;>\;0 .
    \label{eq:cens-unified}
  \end{equation}
\item[(v)] \textup{(Mean version.)} $\E_\mu[\mathrm{Cens}(X)]=\E_\mu[\overline h(X)]-\E_\mu[\underline h(X)]$
  is exactly the width of the sharp identified interval for the population aleatoric entropy
  $\E_\mu[H(p(X))]$ (endpoints $\overline h=H(\mathrm{clip}_{[\plo,\pup]}(\tfrac12))$,
  $\underline h=\min\{H(\plo),H(\pup)\}$, both attained). For every $n$,
  $\E_\mu[\mathrm{Bias}_n(X)]\ge\E_\mu[\mathrm{Cens}(X)]-\E_\mu[\mathrm{Epi}_n(X)]$; if the
  hypothesis of (iii) holds for $\mu$-a.e.\ $x$ then $\E_\mu[\mathrm{Epi}_n(X)]\to0$ and
  $\E_\mu[\mathrm{Bias}_n(X)]\to\E_\mu[\mathrm{Cens}(X)]$, with
  \begin{equation}
    \E_\mu[\mathrm{Cens}(X)]\;\ge\;\E_\mu[\delta(X)\,m(X)]+\tfrac12\E_\mu[\delta(X)^2]
    \;\ge\;\E_\mu[\delta(X)\,m(X)]+\tfrac12\,(\pi_+-\pi_-)^2,
    \label{eq:cens-mean}
  \end{equation}
  $\pi_\pm=\E_\mu[\pup],\E_\mu[\plo]$ the prevalence interval of \cref{thm:sharp-auc}.
\end{enumerate}
\end{corollary}

\begin{proposition}[Plug-in censoring term with cross-fitted nuisances]
\label{prop:cens-plugin}
Let $\hat\plo,\hat\pup$ be the box built from cross-fitted $(\hat e,\hat\eta_1)$ and
$\widehat{\mathrm{Cens}}(x)=\mathrm{EU}([\hat\plo(x),\hat\pup(x)])$. With
$\Delta_n(x)=|\hat e(x)-e(x)|+\tfrac14|\hat\eta_1(x)-\eta_1(x)|$,
\begin{equation}
  \big|\widehat{\mathrm{Cens}}(x)-\mathrm{Cens}(x)\big|\;\le\;2\,H\big(\min\{\Delta_n(x),\tfrac12\}\big)
  \;\le\;2\,\Delta_n(x)\big(1+\log\tfrac1{\Delta_n(x)}\big),
  \label{eq:cens-plugin}
\end{equation}
and, with $\varepsilon_{\mathrm{nuis}}=\|\hat e-e\|_{L^1(\mu)}+\tfrac14\|\hat\eta_1-\eta_1\|_{L^1(\mu)}$
as in \cref{cor:loggamma},
$\E_\mu\big|\widehat{\mathrm{Cens}}(X)-\mathrm{Cens}(X)\big|\le 2H(\min\{\varepsilon_{\mathrm{nuis}},\tfrac12\})$.
Moreover the inflated box $\I^{+}_n(x)=[\hat\plo(x)-\Delta_n(x),\ \hat\pup(x)+\Delta_n(x)]\cap[0,1]$
contains $\I_\Gamma(x)$, so it is valid at $x$ and $\mathrm{EU}(\I^{+}_n(x))\ge\mathrm{Cens}(x)$;
replacing $\Delta_n(x)$ by any radius $r_n(x)\ge\Delta_n(x)$ that holds with probability $1-\alpha$
gives a valid floor at level $1-\alpha$.
\end{proposition}

\begin{proof}[Proof of \cref{cor:uq-full} (and hence of \cref{cor:uq})]
Throughout, $H''(q)=-1/(q(1-q))\le-4$ on $(0,1)$, so for $a,b\in[0,1]$ (by continuity at the
endpoints)
\begin{equation}
  H(a)\;\le\;H(b)+H'(b)(a-b)-2(a-b)^2,\qquad H'(b)=-\logit b .
  \label{eq:sc}
\end{equation}
We also use three elementary facts about $\mathrm{EU}(C)=\max_C H-\min_C H$.
\emph{(F1) Monotonicity:} $C\subseteq C'$ implies $\max_C H\le\max_{C'}H$ and
$\min_C H\ge\min_{C'}H$, hence $\mathrm{EU}(C)\le\mathrm{EU}(C')$.
\emph{(F2) Interval form:} on an interval $[a,b]$, concavity gives
$\max H=H(\mathrm{clip}_{[a,b]}(\tfrac12))$ and $\min H=\min\{H(a),H(b)\}$.
\emph{(F3) Modulus:} for $|a-b|\le\tfrac12$, $|H(a)-H(b)|\le H(|a-b|)$. Indeed, fix
$t=|a-b|$; $u\mapsto H(u+t)-H(u)$ is decreasing on $[0,1-t]$ (its derivative
$H'(u+t)-H'(u)\le0$ by concavity), so it is maximised at $u=0$, giving
$H(u+t)-H(u)\le H(t)-H(0)=H(t)$; the same bound for $H(u)-H(u+t)$ follows from
$H(u)-H(u+t)=H(1-u)-H(1-u-t)$ and the symmetry $H(q)=H(1-q)$. Consequently
$\mathrm{EU}(C)\le H(\min\{\mathrm{diam}\,C,\tfrac12\})$ for every $C$. Also
$H(t)\le t(1+\log\tfrac1t)$ for $t\in(0,\tfrac12]$ because
$g(t)=t+(1-t)\log(1-t)$ has $g(0)=0$, $g'(t)=-\log(1-t)\ge0$.

\emph{(i)} $\log\Gamma>0$ and $\sigma$ strictly increasing give
$\sigma(\eta_1-\log\Gamma)<\sigma(\eta_1)=p_1<\sigma(\eta_1+\log\Gamma)$ (strict because
$\eta_1$ is finite); multiplying by $1-e(x)>0$ and adding $e p_1$ yields
$\plo<p_1<\pup$. The law $P^{(1)}$ with $p_0=p_1$ has tilt $a\equiv1\in[\Gamma^{-1},\Gamma]$,
so it satisfies $\dcsm(\Gamma)$; $P_{\mathrm{obs}}$ depends on $P$ only through
$(P_X,e,p_1)$, which $P^{(1)}$ shares with every admissible $P$. Any statistic
$S(O_1,\dots,O_n)$ is a measurable function of an i.i.d.\ sample from
$P_{\mathrm{obs}}$, so its law is the same under all admissible $P$, and a test of the
hypothesis ``$\mathrm{Epi}_n$ is unbiased for the epistemic uncertainty about $p(x)$''
(true under $P^{(1)}$, false under the tilt $a\equiv\Gamma$ where $p=\pup\ne p_1$) has
power equal to its size.

\emph{(ii)} If $C$ is valid it contains $p(x)$ for every admissible law; by
\cref{lem:box} every $q\in\I_\Gamma(x)$ is $p(x)$ for some admissible law, so
$C\supseteq\I_\Gamma(x)$. Conversely $C\supseteq\I_\Gamma(x)$ is valid because
$p(x)\in\I_\Gamma(x)$ under every admissible law. By (F1),
$\mathrm{EU}(C)\ge\mathrm{EU}(\I_\Gamma(x))=\mathrm{Cens}(x)$, with equality at
$C=\I_\Gamma(x)$. For the reported term: $\bar p_n\in\mathrm{hull}\{p_m\}=C_n$ so
$H(\bar p_n)\le\max_{C_n}H$, and each $H(p_m)\ge\min_{C_n}H$ so
$\tfrac1M\sum_mH(p_m)\ge\min_{C_n}H$; subtracting gives
$\mathrm{Epi}_n\le\mathrm{EU}(C_n)$, and $\mathrm{Epi}_n\ge0$ is Jensen for concave $H$.
Since $\mathrm{hull}(C_n\cup\I_\Gamma)$ contains both $C_n$ and $\I_\Gamma$, (F1) gives
$\mathrm{EU}(\mathrm{hull}(C_n\cup\I_\Gamma))\ge\max\{\mathrm{EU}(C_n),\mathrm{Cens}(x)\}$,
which is \eqref{eq:uq-bias-finite} after subtracting $\mathrm{Epi}_n$.

\emph{(iii)} $C_n\subseteq[p_1-\varepsilon_n,p_1+\varepsilon_n]$ has diameter
$\le2\varepsilon_n$, so by (ii) and (F3)
$\mathrm{Epi}_n\le\mathrm{EU}(C_n)\le H(\min\{2\varepsilon_n,\tfrac12\})\to H(0)=0$.
For the upper bound on the bias, $p_1\in\I_\Gamma(x)$ (part (i)) gives
$\mathrm{hull}(C_n\cup\I_\Gamma)\subseteq[\plo-\varepsilon_n,\pup+\varepsilon_n]\cap[0,1]=:J_n$.
By (F2) $\max_{J_n}H=H(\mathrm{clip}_{J_n}(\tfrac12))$ and
$\max_{\I_\Gamma}H=H(\mathrm{clip}_{\I_\Gamma}(\tfrac12))$; the two clipped points differ by
at most $\varepsilon_n$ (clipping to an interval is $1$-Lipschitz in the endpoints), so by
(F3) the maxima differ by at most $H(\min\{\varepsilon_n,\tfrac12\})$. Likewise
$\min_{J_n}H=\min\{H(\plo-\varepsilon_n),H(\pup+\varepsilon_n)\}$ (endpoints clipped to
$[0,1]$) differs from $\min\{H(\plo),H(\pup)\}$ by at most $H(\min\{\varepsilon_n,\tfrac12\})$,
because $|\min\{u,v\}-\min\{u',v'\}|\le\max\{|u-u'|,|v-v'|\}$. Hence
$\mathrm{EU}(J_n)\le\mathrm{Cens}(x)+2H(\min\{\varepsilon_n,\tfrac12\})$, and (F1) gives the
stated sandwich; both ends converge to $\mathrm{Cens}(x)$, so
$\mathrm{Bias}_n(x)\to\mathrm{Cens}(x)$. For \eqref{eq:uq-overconf}: by (ii) the inner
infimum equals $\mathrm{Cens}(x)-\mathrm{Epi}_n(x)$ for every $n$, and
$\mathrm{Epi}_n(x)\to0$. Positivity of $\mathrm{Cens}(x)$ is (iv).

\emph{(iv)} By the symmetry $H(q)=H(1-q)$ and $\logit(1-q)=-\logit q$ we may assume
$\pup\le\tfrac12$ in the first case. Then $H'=-\logit\ge0$ on $[\plo,\pup]$, so
$\mathrm{Cens}=H(\pup)-H(\plo)=\int_{\plo}^{\pup}|\logit q|\,dq$; since $|\logit|$ is
decreasing on $(0,\tfrac12]$, $m(x)=|\logit\pup|=H'(\pup)$ and
$\max_{\I_\Gamma}|\logit|=|\logit\plo|$, giving the upper bound
$\mathrm{Cens}\le\delta\max|\logit|$ directly from the integral. For the lower bound apply
\eqref{eq:sc} with $a=\plo$, $b=\pup$:
$H(\plo)\le H(\pup)-H'(\pup)\,\delta-2\delta^2$, i.e.\
$\mathrm{Cens}\ge\delta\,m(x)+2\delta^2$, which is \eqref{eq:cens-mono}.
If $\tfrac12\in\I_\Gamma$, (F2) gives $\max H=H(\tfrac12)=\log2$ and
$\min H=H(q^\ast)$ with $|q^\ast-\tfrac12|=d(x)$ (H decreases away from $\tfrac12$). A direct
computation gives $\KL(\mathrm{Bern}(q)\|\mathrm{Bern}(\tfrac12))
=q\log(2q)+(1-q)\log(2(1-q))=\log2-H(q)$, and \eqref{eq:sc} with $b=\tfrac12$, $H'(\tfrac12)=0$,
gives $\log2-H(q^\ast)\ge2d(x)^2$ (equivalently Pinsker with total variation $d(x)$). Since
$\tfrac12\in[\plo,\pup]$, $d(x)\ge\delta/2$, so $2d^2\ge\delta^2/2$; this is
\eqref{eq:cens-inside}. The unified bound \eqref{eq:cens-unified} follows because
$m(x)=0$ in the second case and $2\delta^2\ge\delta^2/2$ in the first. For the closed form
of $\delta$, write $o=\odds(p_1)$; then $\sigma(\eta_1+\log\Gamma)=\Gamma o/(1+\Gamma o)$
and $\sigma(\eta_1-\log\Gamma)=o/(\Gamma+o)$, whose difference is
$o(\Gamma^2-1)/[(1+\Gamma o)(\Gamma+o)]$; multiply by $1-e$. It is $>0$ iff $\Gamma>1$,
$e<1$ and $o\in(0,\infty)$.

\emph{(v)} $\E_\mu[H(p(X))]$ is separable in $p(x)$ and $\I_\Gamma$ is a product of
intervals, so the pointwise extremes of (F2) are attained simultaneously by one admissible
$p$ (measurability: $\overline h,\underline h$ are Borel functions of $(\plo,\pup)$), and the
sharp interval for the mean is $[\E\underline h,\E\overline h]$ with width
$\E[\overline h-\underline h]=\E[\mathrm{Cens}(X)]$. Taking $\E_\mu$ of
\eqref{eq:uq-bias-finite} gives the finite-$n$ mean inequality. Under $\mu$-a.e.\
contraction, $\mathrm{Epi}_n(X)\to0$ a.e.\ and $0\le\mathrm{Epi}_n\le\log2$, so
$\E_\mu[\mathrm{Epi}_n(X)]\to0$ by bounded convergence, and likewise
$\E_\mu[\mathrm{Bias}_n(X)]\to\E_\mu[\mathrm{Cens}(X)]$ from the pointwise sandwich in (iii)
(both bounds are in $[0,2\log2]$). Finally \eqref{eq:cens-mean} is the expectation of
\eqref{eq:cens-unified} followed by Jensen, $\E[\delta^2]\ge(\E\delta)^2$, and
$\E_\mu[\delta]=\E_\mu[\pup]-\E_\mu[\plo]=\pi_+-\pi_-$.
\end{proof}

\begin{proof}[Proof of \cref{prop:cens-plugin}]
Write $\plo(e,\eta)=e\sigma(\eta)+(1-e)\sigma(\eta-\log\Gamma)$ on the convex set
$[0,1]\times\R$. Then $\partial_e\plo=\sigma(\eta)-\sigma(\eta-\log\Gamma)\in[0,1]$ and
$\partial_\eta\plo=e\sigma'(\eta)+(1-e)\sigma'(\eta-\log\Gamma)\le\tfrac14$, and the same
holds for $\pup$; the mean value theorem along the segment from $(e,\eta_1)$ to
$(\hat e,\hat\eta_1)$ gives $|\hat\plo-\plo|,|\hat\pup-\pup|\le\Delta_n(x)$. Now repeat the
perturbation argument of part (iii): the clipped maximisers differ by at most
$\Delta_n$ and the endpoint minima by at most $\Delta_n$, so by (F3) each of
$\max H$ and $\min H$ moves by at most $H(\min\{\Delta_n,\tfrac12\})$, giving
\eqref{eq:cens-plugin}; the second inequality is $H(t)\le t(1+\log\tfrac1t)$. For the mean
bound, $Z=\min\{\Delta_n(X),\tfrac12\}\in[0,\tfrac12]$, $H$ is concave and increasing on
$[0,\tfrac12]$, so $\E H(Z)\le H(\E Z)\le H(\min\{\E\Delta_n,\tfrac12\})$ and
$\E\Delta_n\le\varepsilon_{\mathrm{nuis}}$. Finally $|\hat\plo-\plo|\le\Delta_n$ gives
$\hat\plo-\Delta_n\le\plo$ and $\hat\pup+\Delta_n\ge\pup$, so
$\I^{+}_n(x)\supseteq\I_\Gamma(x)$, and validity plus (F1) give
$\mathrm{EU}(\I^{+}_n(x))\ge\mathrm{Cens}(x)$; with a probabilistic radius the same holds on
the event $\{r_n(x)\ge\Delta_n(x)\}$.
\end{proof}
